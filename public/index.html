การใช้งานแบบ Production ต้องตั้งค่า PostgreSQL และคีย์เข้ารหัสใน AWS Amplify Environment Variables

DATABASE_URLSESSION_SECRETDATA_KEYLOCAL_WORKSPACE_DIR=/tmp/ai-code-chat-workspacesLU_CHAT_BASE_URLLU_CHAT_TOKENLU_COMFY_BASE_URL
